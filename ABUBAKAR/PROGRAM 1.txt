  <h2>Develop a Javascript Program to Capitalize the first letter of each Word of a sentence (String).</h2>
  <script>
    function capitalizeFirstLetterOfEachWord(sentence) {
      return sentence.split(' ').map(word => {
        return word.charAt(0).toUpperCase() + word.slice(1).toLowerCase();
      }).join(' ');
    }

    const inputString = "great. you have successfully executed the program";
    const capitalizedString = capitalizeFirstLetterOfEachWord(inputString);
    document.write('<h3>' + capitalizedString + '</h3>');
  </script>
